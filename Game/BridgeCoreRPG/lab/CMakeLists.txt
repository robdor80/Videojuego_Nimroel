cmake_minimum_required(VERSION 3.25)
project(NimroelCoreRPGLab LANGUAGES CXX)
include(../cmake/dependencies.cmake)
add_library(nimroel_core_lab SHARED src/core_rpg_lab.cpp)
target_link_libraries(nimroel_core_lab PRIVATE CoreRPG::CoreRPG godot::cpp)
set_target_properties(nimroel_core_lab PROPERTIES
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/project/bin"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/project/bin"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/project/bin")
if(MSVC)
    target_link_options(nimroel_core_lab PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/nimroel.map")
else()
    target_link_options(nimroel_core_lab PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/nimroel.map")
endif()
if(ANDROID)
    set_target_properties(nimroel_core_lab PROPERTIES OUTPUT_NAME nimroel_core_lab.android.arm64)
    target_link_options(nimroel_core_lab PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
elseif(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    set_target_properties(nimroel_core_lab PROPERTIES OUTPUT_NAME nimroel_core_lab.windows.x86_64)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
    set_target_properties(nimroel_core_lab PROPERTIES OUTPUT_NAME nimroel_core_lab.linux.x86_64)
    target_link_options(nimroel_core_lab PRIVATE -Wl,--no-undefined)
else()
    message(FATAL_ERROR "Unsupported laboratory platform")
endif()
