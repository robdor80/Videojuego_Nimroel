cmake_minimum_required(VERSION 3.25)
project(EdranorE11 LANGUAGES CXX)
if(NOT CORERPG_ROOT)
  message(FATAL_ERROR "Please set CORERPG_ROOT")
endif()
set(BUILD_TESTING OFF CACHE BOOL "No upstream full suite" FORCE)
set(CORERPG_BUILD_CONTENT_JSON ON CACHE BOOL "Official JSON parser" FORCE)
add_subdirectory("${CORERPG_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/corerpg" EXCLUDE_FROM_ALL)
add_executable(edranor_e11_core_smoke cpp_smoke.cpp)
target_link_libraries(edranor_e11_core_smoke PRIVATE CoreRPG::ContentJson)
target_compile_features(edranor_e11_core_smoke PRIVATE cxx_std_20)
enable_testing()
add_test(NAME edranor_e11_core_catalog_smoke COMMAND edranor_e11_core_smoke "${CMAKE_CURRENT_SOURCE_DIR}/../../ContentPacks/edranor-territory/pack")
